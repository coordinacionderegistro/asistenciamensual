-- ESQUEMA COMPLETO BASE DE DATOS: politecnico_asistencia_db
-- Compatible con MySQL 8.0, PostgreSQL, Supabase y SQLite

CREATE TABLE roles (
    role_id INT AUTO_INCREMENT PRIMARY KEY,
    nombre_rol VARCHAR(50) NOT NULL UNIQUE
);

INSERT INTO roles (nombre_rol) VALUES ('Administrador'), ('Docente Admin'), ('Docente'), ('Padre');

CREATE TABLE usuarios (
    usuario_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    nombre_completo VARCHAR(100) NOT NULL,
    role_id INT NOT NULL,
    FOREIGN KEY (role_id) REFERENCES roles(role_id)
);

CREATE TABLE estudiantes (
    estudiante_id INT AUTO_INCREMENT PRIMARY KEY,
    rne VARCHAR(20) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    sexo CHAR(1) DEFAULT 'M',
    familia VARCHAR(100),
    salida VARCHAR(150),
    grado VARCHAR(10),
    seccion CHAR(1)
);

CREATE TABLE asistencias_mensuales (
    asistencia_id INT AUTO_INCREMENT PRIMARY KEY,
    estudiante_id INT NOT NULL,
    mes VARCHAR(10) NOT NULL,
    dias_asistidos INT DEFAULT 0,
    FOREIGN KEY (estudiante_id) REFERENCES estudiantes(estudiante_id) ON DELETE CASCADE
);
